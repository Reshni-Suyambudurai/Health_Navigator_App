// Patient AI Health Navigator — MVP database schema (DBML)
// Review notes:
// - Enum-like varchar fields list the currently proposed allowed values in comments; enforce with DB ENUM/CHECK/lookup tables in implementation.
// - Clinical guardrails must be deterministically enforced in application logic; Markdown is supporting guidance, not executable policy by itself.
// - FHIR remains the clinical source of truth; store external FHIR identifiers rather than duplicating all clinical records here.
// - This MVP uses one primary role per user. If staff can hold multiple roles, add a user_roles junction table.
// - Use UTC timestamps consistently and apply organization/patient-level authorization to all clinical records.
 
 
Table organizations {
  organization_id bigint [primary key]
  organization_name varchar
  organization_type varchar // HOSPITAL, CLINIC, HEALTHCARE_GROUP, PAYER, ACO
  npi_number varchar
  address varchar
  phone varchar
  status varchar // ACTIVE, INACTIVE
  created_at timestamp
  updated_at timestamp
}
 
Table roles {
  role_id bigint [primary key]
  role_name varchar // PATIENT, CARE_MANAGER, TRANSITIONAL_CARE_NURSE, PHARMACIST, SOCIAL_WORKER, POPULATION_HEALTH_MANAGER, CLINICAL_GOVERNANCE, SYSTEM_ADMIN
  description varchar
  created_at timestamp
}
 
Table users {
  user_id bigint [primary key]
  organization_id bigint
  role_id bigint
  first_name varchar
  last_name varchar
  email varchar
  password_hash varchar
  phone varchar
  status varchar // ACTIVE, INACTIVE, SUSPENDED
  is_verified boolean
  last_login_at timestamp
  created_at timestamp
  updated_at timestamp
}
 
Table patient_profiles {
  patient_id bigint [primary key]
  user_id bigint
  external_fhir_patient_id varchar
  date_of_birth date
  gender varchar // FEMALE, MALE, NON_BINARY, OTHER, UNKNOWN, NOT_DISCLOSED (align with FHIR/organization policy)
  preferred_language varchar
  preferred_contact_method varchar // SMS, EMAIL, PHONE, PORTAL
  emergency_contact_name varchar
  emergency_contact_phone varchar
  risk_tier varchar // LOW, MEDIUM, HIGH, CRITICAL, UNKNOWN
  created_at timestamp
  updated_at timestamp
}
 
Table staff_profiles {
  staff_id bigint [primary key]
  user_id bigint
  employee_id varchar
  license_number varchar
  license_type varchar
  department varchar
  specialization varchar
  created_at timestamp
  updated_at timestamp
}
 
Ref: users.organization_id > organizations.organization_id
Ref: users.role_id > roles.role_id
Ref: patient_profiles.user_id > users.user_id
Ref: staff_profiles.user_id > users.user_id
 
 
Table patient_consent_records {
  consent_id bigint [primary key]
  patient_id bigint
  consent_type varchar // AI_DISCLOSURE, AI_INTERACTION, OUTREACH, DATA_USE
  consent_status varchar // ACCEPTED, DECLINED, REVOKED, PENDING
  consent_text_version varchar
  effective_at timestamp
  revoked_at timestamp
  captured_channel varchar // PORTAL, SMS, PHONE, IN_PERSON
  captured_by_user_id bigint
  created_at timestamp
}
 
Table patient_enrollments {
  enrollment_id bigint [primary key]
  patient_id bigint
  organization_id bigint
  enrollment_program varchar // TCM, APCM, RPM, CHRONIC_CARE
  enrollment_status varchar // ACTIVE, INACTIVE, COMPLETED, WITHDRAWN
  enrollment_date timestamp
  close_date timestamp
  closed_reason varchar
  created_by_user_id bigint
  created_at timestamp
  updated_at timestamp
}
 
Table care_navigation_episodes {
  episode_id bigint [primary key]
  patient_id bigint
  enrollment_id bigint
  episode_type varchar // POST_DISCHARGE_30_DAY, MEDICATION_SUPPORT, CHRONIC_CARE_FOLLOWUP, APPOINTMENT_FOLLOWUP, GENERAL_NAVIGATION
  trigger_type varchar // DISCHARGE, PATIENT_REQUEST, MEDICATION_CONCERN, SCHEDULED_OUTREACH, STAFF_REFERRAL
  trigger_reference varchar
  status varchar // OPEN, IN_PROGRESS, ON_HOLD, COMPLETED, CLOSED
  opened_at timestamp
  closed_at timestamp
  owner_staff_id bigint
  created_at timestamp
  updated_at timestamp
}
 
Ref: patient_consent_records.patient_id > patient_profiles.patient_id
Ref: patient_consent_records.captured_by_user_id > users.user_id
Ref: patient_enrollments.patient_id > patient_profiles.patient_id
Ref: patient_enrollments.organization_id > organizations.organization_id
Ref: patient_enrollments.created_by_user_id > users.user_id
Ref: care_navigation_episodes.patient_id > patient_profiles.patient_id
Ref: care_navigation_episodes.enrollment_id > patient_enrollments.enrollment_id
Ref: care_navigation_episodes.owner_staff_id > staff_profiles.staff_id
 
 
Table conversation_threads {
  thread_id bigint [primary key]
  patient_id bigint
  episode_id bigint
  channel varchar // SMS, WEB, PORTAL, PHONE
  language varchar // BCP-47 language tag, e.g. en-US; not a fixed enum
  started_at timestamp
  ended_at timestamp
  status varchar // ACTIVE, COMPLETED, ABANDONED
  created_at timestamp
}
 
Table conversation_messages {
  message_id bigint [primary key]
  thread_id bigint
  sender_type varchar // PATIENT, AI, STAFF, SYSTEM
  sender_user_id bigint
  ai_model_name varchar
  ai_model_version varchar
  message_text text
  message_intent varchar // e.g. APPOINTMENT_QUESTION, MEDICATION_CONCERN, SYMPTOM_REPORT, DISCHARGE_INSTRUCTIONS, HUMAN_HELP_REQUEST, OTHER; preferably controlled taxonomy
  readability_grade integer // Optional readability score/grade; define the scoring method if used
  is_guardrail_override boolean // true only when an authorized override is explicitly applied
  message_metadata json
  sent_at timestamp
}
 
Table clinical_guardrail {
  guardrail_id bigint [primary key]
  guardrail_name varchar
  guardrail_markdown text
  created_at timestamp
  updated_at timestamp
}
 
Table clinical_guardrail_rules {
  rule_id bigint [primary key]
  guardrail_id bigint
  rule_name varchar
  rule_category varchar // EMERGENCY, MEDICATION, SYMPTOM, PRIVACY, CLINICAL_SCOPE
  urgency_level varchar // ROUTINE, SAME_DAY, EMERGENCY
  handler_role_id bigint // Default handling role; nullable if not applicable
  action_type varchar // SAFE_RESPOND, BLOCK_RESPONSE, ESCALATE, EMERGENCY_PROTOCOL
  status varchar // DRAFT, ACTIVE, INACTIVE, RETIRED
  patient_response_template text // Approved patient-safe response; used instead of the LLM when matched
  target_queue_type varchar // MEDICATION, CLINICAL, EMERGENCY, APPOINTMENT, SOCIAL_SUPPORT
  escalation_type varchar // MEDICATION, CLINICAL_REVIEW, EMERGENCY, APPOINTMENT, SOCIAL_SUPPORT
  escalation_priority varchar // LOW, MEDIUM, HIGH, CRITICAL
  created_at timestamp
  updated_at timestamp
}
 
Ref: clinical_guardrail_rules.guardrail_id > clinical_guardrail.guardrail_id
Ref: clinical_guardrail_rules.handler_role_id > roles.role_id

Table clinical_guardrail_matchers {
  matcher_id bigint [primary key]
  rule_id bigint
  pattern varchar // Normalized deterministic phrase or | separated ALL_TERMS condition
  match_type varchar // PHRASE, ALL_TERMS
  priority integer // Lower number evaluates first within the same urgency/category
  is_active boolean
  created_at timestamp
  updated_at timestamp
}

Ref: clinical_guardrail_matchers.rule_id > clinical_guardrail_rules.rule_id
 
Table conversation_guardrail_hits {
  guardrail_hit_id bigint [primary key]
  thread_id bigint
  message_id bigint
  patient_id bigint
  rule_id bigint
  hit_urgency_level varchar // ROUTINE, SAME_DAY, EMERGENCY
  action_taken varchar // SAFE_RESPOND, BLOCK_RESPONSE, ESCALATE, EMERGENCY_PROTOCOL
  hit_reason text
  created_at timestamp
}
 
Ref: conversation_guardrail_hits.thread_id > conversation_threads.thread_id
Ref: conversation_guardrail_hits.message_id > conversation_messages.message_id
Ref: conversation_guardrail_hits.patient_id > patient_profiles.patient_id
Ref: conversation_guardrail_hits.rule_id > clinical_guardrail_rules.rule_id
 
Table emergency_events {
  emergency_event_id bigint [primary key]
  patient_id bigint
  thread_id bigint
  guardrail_hit_id bigint
  emergency_type varchar // CHEST_PAIN, BREATHING_DIFFICULTY, STROKE_SIGNS, SELF_HARM_RISK, OTHER; define approved clinical categories
  emergency_instruction_sent boolean
  notified_care_team boolean
  event_status varchar // DETECTED, ESCALATED, ACKNOWLEDGED, RESOLVED
  detected_at timestamp
  resolved_at timestamp
  created_at timestamp
}
 
Ref: conversation_threads.patient_id > patient_profiles.patient_id
Ref: conversation_threads.episode_id > care_navigation_episodes.episode_id
Ref: conversation_messages.thread_id > conversation_threads.thread_id
Ref: conversation_messages.sender_user_id > users.user_id
 
 
Table care_team_queues {
  queue_id bigint [primary key]
  organization_id bigint
  queue_name varchar
  queue_type varchar // NURSE, PHARMACIST, CARE_MANAGER, SOCIAL_WORKER
  sla_minutes integer
  status varchar // ACTIVE, INACTIVE
  created_at timestamp
}
 
Table queue_memberships {
  queue_membership_id bigint [primary key]
  queue_id bigint
  staff_id bigint
  role_in_queue varchar // PRIMARY, SECONDARY, SUPERVISOR
  is_active boolean
  created_at timestamp
}
 
Table escalations {
  escalation_id bigint [primary key]
  patient_id bigint
  episode_id bigint
  thread_id bigint
  guardrail_hit_id bigint
  queue_id bigint
  escalation_type varchar // EMERGENCY, MEDICATION, CLINICAL, APPOINTMENT, SOCIAL, OPERATIONAL
  priority varchar // LOW, MEDIUM, HIGH, CRITICAL
  reason text
  patient_summary text
  status varchar // OPEN, ASSIGNED, IN_PROGRESS, WAITING, RESOLVED, CLOSED
  created_at timestamp
  resolved_at timestamp
}
 
Table escalation_assignments {
  assignment_id bigint [primary key]
  escalation_id bigint
  assigned_to_staff_id bigint
  assigned_by_user_id bigint
  assigned_at timestamp
  unassigned_at timestamp
  is_active boolean
}
 
Table escalation_actions {
  escalation_action_id bigint [primary key]
  escalation_id bigint
  action_by_user_id bigint
  action_type varchar // REVIEW, CALL_ATTEMPT, PATIENT_CONTACTED, MESSAGE_SENT, REFERRED, RESOLVED, OTHER
  action_notes text
  action_at timestamp
}

  ip_address varchar
  correlation_id varchar
  created_at timestamp
}
